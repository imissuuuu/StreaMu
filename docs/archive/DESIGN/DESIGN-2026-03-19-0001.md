```cpp
static std::string parse_string(const std::string& json, const char* key,
                                 const std::string& default_val) {
    std::string search = std::string("\"") + key + "\"";
    size_t pos = json.find(search);
    if (pos == std::string::npos) return default_val;
    pos = json.find(':', pos);
    if (pos == std::string::npos) return default_val;
    pos++;
    while (pos < json.size() && (json[pos] == ' ' || json[pos] == '\t')) pos++;
    if (pos >= json.size() || json[pos] != '"') return default_val;
    pos++; // 開き引用符をスキップ
    std::string result;
    while (pos < json.size() && json[pos] != '"') result += json[pos++];
    return result;
}
```

バッファサイズを 512→1024 に拡張（文字列フィールド追加のため）。

#### (B) アクション文字列 `trigger_fav` ハンドラ追加

既存の `trigger_playing` ハンドラの後に追加：

```cpp
} else if (action == "trigger_fav") {
    if (ctx.config.fav_playlist_id.empty()) {
        // 未登録 → プレイリスト一覧へ
        screen_mgr.change_screen(ctx, "PlaylistsScreen");
    } else {
        // FAVプレイリストを探してシャッフル再生
        LightLock_Lock(&ctx.lock);
        bool found = false;
        for (const auto& pl : ctx.playlists) {
            if (pl.id == ctx.config.fav_playlist_id) {
                ctx.selected_playlist_id = pl.id;
                ctx.g_tracks             = pl.tracks;
                ctx.selected_index       = 0;
                found = true;
                break;
            }
        }
        LightLock_Unlock(&ctx.lock);
        if (found && !ctx.g_tracks.empty()) {
            start_playback(ctx.g_tracks[0]);
            LightLock_Lock(&ctx.lock);
            ctx.active_playlist_id = ctx.config.fav_playlist_id;
            for (const auto& pl : ctx.playlists) {
                if (pl.id == ctx.config.fav_playlist_id) {
                    ctx.active_playlist_name = pl.name;
                    break;
                }
            }
            ctx.playing_tracks = ctx.g_tracks;
            ctx.play_queue.clear();
            for (size_t i = 0; i < ctx.playing_tracks.size(); ++i)
                ctx.play_queue.push_back(i);
            // Fisher-Yates shuffle
            for (size_t i = ctx.play_queue.size() - 1; i > 0; --i) {
                size_t j = rand() % (i + 1);
                std::swap(ctx.play_queue[i], ctx.play_queue[j]);
            }
            ctx.current_track_idx = 0;
            LightLock_Unlock(&ctx.lock);
            screen_mgr.change_screen(ctx, "PlayingScreen");
        }
    }
    kDown &= ~KEY_A;
```

## 4. 各機能の詳細設計

### FR-1/FR-2: FAVボタン → シャッフル再生 or プレイリスト一覧

```
タップ/A押下
  └─ get_action_for_index("Favorites") → "trigger_fav"
       ├─ fav_playlist_id が空
       │    └─ change_screen("PlaylistsScreen")
       └─ fav_playlist_id が設定済み
            ├─ playlists から該当PL を検索
            ├─ 見つかった → シャッフル再生 → PlayingScreen
            └─ 見つからない (削除済み等) → 何もしない
```

### FR-3/FR-4: SELECTメニュー拡張

```
KEY_SELECT (STATE_PLAYLISTS)
  └─ STATE_POPUP_PLAYLIST_OPTIONS に遷移
       [0] タイトルを編集
       [1] ☆ お気に入り登録  ← fav_playlist_id != selected_playlist_id の場合
           ★ お気に入り解除  ← fav_playlist_id == selected_playlist_id の場合
       [2] 削除
```
