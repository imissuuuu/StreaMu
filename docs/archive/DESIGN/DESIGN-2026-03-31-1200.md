### FR-4: エージェントループ

```python
def run_agent(question, port):
    messages = [{"role": "user", "content": QUERY_TEMPLATE.format(Question=question)}]
    tool_log = []

    for i in range(MAX_LOOPS):
        resp = call_qwen(messages, port, use_tools=True)
        choice = resp["choices"][0]

        # ツール呼び出しなし → 回答確定
        if choice["finish_reason"] != "tool_calls":
            return choice["message"]["content"], tool_log

        # ツール実行
        messages.append(choice["message"])
        for tc in choice["message"]["tool_calls"]:
            name = tc["function"]["name"]
            args = json.loads(tc["function"]["arguments"])
            print(f"  [Loop {i+1}/{MAX_LOOPS}] {name}({args})")
            result = execute_tool(name, args)
            tool_log.append(name)
            messages.append({
                "role": "tool",
                "tool_call_id": tc["id"],
                "content": result
            })

    # ループ上限到達 → 強制回答
    messages.append({"role": "user", "content": "Provide your final answer now."})
    resp = call_qwen(messages, port, use_tools=False)
    return resp["choices"][0]["message"]["content"], tool_log
```

### FR-2: Qwen呼び出し

```python
def call_qwen(messages, port, use_tools=True):
    payload = {
        "model": "qwen",
        "messages": messages,
        "max_tokens": 2048,
        "temperature": 0.1,
        "chat_template_kwargs": {"enable_thinking": False},
    }
    if use_tools:
        payload["tools"] = TOOLS
        payload["tool_choice"] = "auto"

    data = json.dumps(payload).encode()
    req = urllib.request.Request(
        QWEN_URL.format(port=port),
        data=data,
        headers={"Content-Type": "application/json"}
    )
    with urllib.request.urlopen(req, timeout=120) as r:
        return json.loads(r.read())
```

def browse_url(url: str) -> str:
    env = {**os.environ, "PYTHONIOENCODING": "utf-8"}
    base = "source ~/.bashrc && "
    subprocess.run(
        ["bash", "-c", f'{base}browser-use open "{url}"'],
        env=env, capture_output=True, timeout=30
    )
    r = subprocess.run(
        ["bash", "-c", f'{base}browser-use eval "document.body.innerText"'],
        env=env, capture_output=True, text=True, timeout=30
    )
    return r.stdout[:BROWSE_CONTENT_LIMIT]
